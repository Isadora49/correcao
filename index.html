import { PDFDocument, PDFName, PDFString } from 'pdf-lib'; import * as fs from 'fs'; async function injetarCampoImagem() { // 1. Carrega o arquivo PDF base const pdfExistenteBytes = fs.readFileSync('./fichas/ficha_base.pdf'); const pdfDoc = await PDFDocument.load(pdfExistenteBytes); // Pega a primeira página da ficha (index 0) const paginas = pdfDoc.getPages(); const primeiraPagina = paginas[0]; // 2. Pega o formulário do PDF const formulario = pdfDoc.getForm(); // 3. Cria o botão que servirá de campo para a foto do personagem const nomeDoCampo = 'BotaoAvatarPersonagem'; const botaoFoto = formulario.createButton(nomeDoCampo); // Define a posição e o tamanho do quadrado da foto (Altere os valores conforme sua ficha) botaoFoto.addToPage(primeiraPagina, { x: 50, // Posição horizontal (esquerda para direita) y: 600, // Posição vertical (baixo para cima) width: 120, // Largura do quadro da foto height: 150 // Altura do quadro da foto }); // 4. INJEÇÃO DO JAVASCRIPT DO ACROBAT (O Pulo do Gato) // Criamos a ação em JavaScript que o Adobe Reader vai executar ao clicar no botão const acaoJavaScript = pdfDoc.context.obj({ S: 'JavaScript', JS: PDFString.of( `try { this.importIcon("${nomeDoCampo}"); } catch(e) { app.alert("Para adicionar uma foto, abra esta ficha usando o Adobe Acrobat Reader no Computador."); }` ), }); // Associamos essa ação ao evento "A" (Ação/Clique) do botão botaoFoto.acroField.dict.set(PDFName.of('A'), acaoJavaScript); // 5. Salva o novo PDF modificado const pdfFinalBytes = await pdfDoc.save(); fs.writeFileSync('./ficha_rpg_final.pdf', pdfFinalBytes); console.log('✅ Ficha gerada com sucesso com campo de imagem editável!'); } injetarCampoImagem().catch(err => console.error('Erro ao gerar ficha:', err));
